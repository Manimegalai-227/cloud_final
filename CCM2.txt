n = 10

if rank == 0:
    if n % 2 == 0:
        print(n, "is Even")
    else:
        print(n, "is Odd")
factorial :

from mpi4py import MPI

comm = MPI.COMM_WORLD
rank = comm.Get_rank()

n = 5
fact = 1

if rank == 0:
    for i in range(1, n + 1):
        fact *= i

    print("Factorial =", fact)

int main() {
    #pragma omp parallel
    {
        cout << "Hello World from thread "

    }

}

file name: nano reduction.cpp
#include <iostream>
#include <omp.h>
using namespace std;

int main() {
    int sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= 10; i++) {
        sum += i;
    }

    cout << "Sum = " << sum << endl;
    return 0;
}
run:
g++ -fopenmp reduction.cpp -o reduction
./reduction
