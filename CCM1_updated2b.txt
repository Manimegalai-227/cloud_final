const express = require('express');

const app = express();

app.use(express.json());

let books = [
    { title: 'Harry Potter', id: 1 },
    { title: 'Twilight', id: 2 },
    { title: 'Lorien Legacies', id: 3 }

app.get('/', (req, res) => {
    res.send('REST APIs are so much fun');
});

// GET - Get all books
app.get('/api/books', (req, res) => {
    res.send(books);
});

// GET - Get book by ID
app.get('/api/books/:id', (req, res) => {
    const book = books.find(
        c => c.id === parseInt(req.params.id)
    );

    if (!book) {
        return res.status(404).send(
            "Oops... Can't find what you are looking for!"
        );
    }

    res.send(book);
});

// POST - Add new book
app.post('/api/books', (req, res) => {
    const book = {
        title: req.body.title,
        id: books.length > 0
            ? books[books.length - 1].id + 1
            : 1
    };

    books.push(book);

    res.status(201).send(book);
});

// PUT - Update book
app.put('/api/books/:id', (req, res) => {
    const book = books.find(
        c => c.id === parseInt(req.params.id)
    );

    if (!book) {
        return res.status(404).send('Book not found');
    }

    book.title = req.body.title;

    res.send(book);
});

// DELETE - Delete book
app.delete('/api/books/:id', (req, res) => {
    const id = parseInt(req.params.id);

    const book = books.find(c => c.id === id);

    if (!book) {
        return res.status(404).send('Book not found');
    }

    books = books.filter(c => c.id !== id);

    res.send(book);
});

// START SERVER
const port = process.env.PORT || 8080;

app.listen(port, () => {
    console.log(`Listening on port ${port}...`);
});

App.jsx


import { useEffect, useState } from "react";
import { auth, db } from "./firebase";
import { GoogleAuthProvider, signInWithPopup, signOut } from "firebase/auth";
import { collection, addDoc, getDocs, updateDoc, deleteDoc, doc } from "firebase/firestore";

function App() {
  const [user, setUser] = useState(null);
  const [students, setStudents] = useState([]);
  const [name, setName] = useState("");
  const [age, setAge] = useState("");
  const [department, setDepartment] = useState("");

  const login = async () => {
    const provider = new GoogleAuthProvider();
    try {
      const result = await signInWithPopup(auth, provider);
      setUser(result.user);
    } catch (error) {
      console.log(error);
    }
  };

  const logout = async () => {
    await signOut(auth);
    setUser(null);
  };

  const addStudent = async () => {
    await addDoc(collection(db, "students"), {
      name: name,
      age: Number(age),
      department: department
    });
    setName("");
    setAge("");
    setDepartment("");
    getStudents();
  };

  const getStudents = async () => {
    const querySnapshot = await getDocs(collection(db, "students"));
    const data = querySnapshot.docs.map((document) => ({
      id: document.id,
      ...document.data()
    }));
    setStudents(data);
  };

  const updateStudent = async (id) => {
    await updateDoc(doc(db, "students", id), {
      department: "CSE"
    });
    getStudents();
  };

  const deleteStudent = async (id) => {
    await deleteDoc(doc(db, "students", id));
    getStudents();
  };

  useEffect(() => {
    getStudents();
  }, []);

  return (
    <div>
      <h1>Student Management System</h1>
      {!user ? (
        <button onClick={login}>Login with Google</button>
      ) : (
        <div>
          <h3>Welcome {user.displayName}</h3>
          <button onClick={logout}>Logout</button>
        </div>
      )}
      <hr />
      <h2>Add Student</h2>
      <input placeholder="Name" value={name} onChange={(e) => setName(e.target.value)} />
      <input placeholder="Age" value={age} onChange={(e) => setAge(e.target.value)} />
      <input placeholder="Department" value={department} onChange={(e) => setDepartment(e.target.value)} />
      <button onClick={addStudent}>Add Student</button>
      <h2>Students</h2>
      <button onClick={getStudents}>Refresh</button>
      {students.map((student) => (
        <div key={student.id}>
          <p>{student.name} | {student.age} | {student.department}</p>
          <button onClick={() => updateStudent(student.id)}>Update</button>
          <button onClick={() => deleteStudent(student.id)}>Delete</button>
        </div>
      ))}
    </div>
  );
}

export default App;

async function initializeAuth() {
    const status = document.getElementById("status");

    try {
        if (!window.auth0) {
            throw new Error("Auth0 SDK did not load. Check your internet connection.");
        }

        auth0Client = await window.auth0.createAuth0Client({
            domain: "dev-ylb65zhzxc87dfi1.us.auth0.com",
            clientId: "XptX9H1rbTNNAgrU1FbvzAcynRpD3xVj",
            authorizationParams: {
                redirect_uri: window.location.origin
            }
        });

        // Handle the return from Auth0 after login
        if (
            window.location.search.includes("code=") &&
            window.location.search.includes("state=")
        ) {
            await auth0Client.handleRedirectCallback();

            window.history.replaceState(
                {},
                document.title,
                window.location.pathname
            );
        }

        await updateUI();

    } catch (error) {
        console.error("Auth0 initialization failed:", error);
        status.textContent = "Error: " + error.message;
    }
}

async function updateUI() {
    const loggedIn = await auth0Client.isAuthenticated();

    document.getElementById("loginBtn").hidden = loggedIn;
    document.getElementById("signupBtn").hidden = loggedIn;
    document.getElementById("logoutBtn").hidden = !loggedIn;
    document.getElementById("profile").hidden = !loggedIn;

    if (loggedIn) {
        const user = await auth0Client.getUser();

        document.getElementById("name").textContent =
            user.name || "User";

        document.getElementById("email").textContent =
            user.email || "Email unavailable";

        const avatar = document.getElementById("avatar");

        if (user.picture) {
            avatar.src = user.picture;
            avatar.hidden = false;
        } else {
            avatar.hidden = true;
        }

        document.getElementById("status").textContent =
            "Login successful! Welcome to SecureWeb.";
    } else {
        document.getElementById("status").textContent =
            "Please log in or sign up to continue.";
    }
}

document.getElementById("loginBtn").addEventListener("click", async () => {
    try {
        await auth0Client.loginWithRedirect();
    } catch (error) {
        console.error(error);
        document.getElementById("status").textContent =
            "Login failed: " + error.message;
    }
});

document.getElementById("signupBtn").addEventListener("click", async () => {
    try {
        await auth0Client.loginWithRedirect({
            authorizationParams: {
                screen_hint: "signup"
            }
        });
    } catch (error) {
        console.error(error);
        document.getElementById("status").textContent =
            "Signup failed: " + error.message;
    }
});

document.getElementById("logoutBtn").addEventListener("click", () => {
    auth0Client.logout({
        logoutParams: {
            returnTo: window.location.origin
        }
    });
});

from flask import Flask, render_template, request, redirect, url_for
from pymongo import MongoClient
from bson.objectid import ObjectId

app = Flask(__name__)

# Connect to MongoDB Atlas
uri = "mongodb+srv://manimegalai:12345@cluster0.c0nsppg.mongodb.net/"
client = MongoClient(uri)

db = client["student"]
records = db["student_table"]


# READ: Display all students
@app.route("/")
def index():
    students = list(records.find())
    return render_template("index.html", students=students)


# CREATE: Add a student
@app.route("/add", methods=["POST"])
def add_student():
    student = {
        "name": request.form["name"],
        "email": request.form["email"],
        "phone": request.form["phone"]
    }

    records.insert_one(student)
    return redirect(url_for("index"))


# UPDATE: Change student details
@app.route("/update/<id>", methods=["POST"])
def update_student(id):
    records.update_one(
        {"_id": ObjectId(id)},
        {"$set": {
            "name": request.form["name"],
            "email": request.form["email"],
            "phone": request.form["phone"]
        }}
    )
    return redirect(url_for("index"))


# DELETE: Remove a student
@app.route("/delete/<id>", methods=["POST"])
def delete_student(id):
    records.delete_one({"_id": ObjectId(id)})
    return redirect(url_for("index"))


if __name__ == "__main__":
    app.run(debug=True)
